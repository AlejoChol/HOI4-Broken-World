country_event = { 
	id = rmap.2
	fire_only_once = no
	hidden = yes
	is_triggered_only = yes
	immediate = {
		every_owned_state = {
			limit = {has_state_flag = rmap_owned} 
			every_neighbor_state = {
				limit = {
					NOT = {has_state_flag = rmap_owned}} 
				set_temp_variable = {rmap_rcountry = ROOT }
				log = "[?rmap_rcountry], spreading from state [?PREV.GetID], has acquired state [?THIS.GetID]"
				set_state_flag = rmap_owned 
				ROOT = {transfer_state = PREV} 
				add_core_of = ROOT
			}
		}
	}
}
